{{- if .Values.ingress.enabled }}
Access the UI at: http{{ if include "porter-galaxy.ingressTLS" . }}s{{ end }}://{{ .Values.ingress.host | default "<ingress load balancer address>" }}
{{- else if eq .Values.service.type "LoadBalancer" }}
Get the external IP by running:
  kubectl get svc {{ include "porter-galaxy.fullname" . }}-frontend -n {{ .Release.Namespace }} -w

Then open http://<EXTERNAL-IP>
{{- else }}
Port-forward to access the UI locally:
  kubectl port-forward svc/{{ include "porter-galaxy.fullname" . }}-frontend 8080:80 -n {{ .Release.Namespace }}

Then open http://localhost:8080
{{- end }}

{{- else }}
